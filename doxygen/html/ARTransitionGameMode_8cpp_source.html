50void AARTransitionGameMode::BeginPlay()
51{
52 Super::BeginPlay();
53
54 if (!HasAuthority())
55 {
56 return;
57 }
58
59 bFactionElectionFinalizedForThisTransition = false;
60 bWorldDayAdvancedForThisTransition = false;
61
62 if (UWorld* World = GetWorld())
63 {
64 for (FConstPlayerControllerIterator It = World->GetPlayerControllerIterator(); It; ++It)
65 {
66 EnterTransitionSpectatorState(It->Get());
67 }
68 }
69
70 InitializeTransitionContext();
71 if (bResetPlayerReadyOnBeginPlay)
72 {
73 ResetPlayersReadyState();
74 }
75
76 if (AARTransitionGameState* TransitionGameState = GetGameState<AARTransitionGameState>())
77 {
78 TransitionGameState->OnAllPlayersTravelReadyChanged.AddUniqueDynamic(this, &AARTransitionGameMode::HandleAllPlayersTravelReadyChanged);
79 }
80
81 if (bAutoAdvanceWhenAllPlayersReady)
82 {
83 TryAdvanceToDestination();
84 }
85}

168void AARTransitionGameMode::TryFinalizeFactionElectionFromTransitionContext(const FARTransitionContext& TransitionContext)
169{
170 if (!HasAuthority() || bFactionElectionFinalizedForThisTransition || !bFinalizeFactionElectionOnShopToInvaderTransition)
171 {
172 return;
173 }
174
175 if (TransitionContext.SourceMode != EARTransitionSourceMode::Shop
176 || TransitionContext.Reason != EARTransitionReason::ShopToInvader)
177 {
178 return;
179 }
180
181 UGameInstance* GameInstance = GetGameInstance();
182 UARFactionVotingSubsystem* VotingSubsystem = GameInstance ? GameInstance->GetSubsystem<UARFactionVotingSubsystem>() : nullptr;
183 if (!VotingSubsystem)
184 {
185 UE_LOG(ARLog, Warning, TEXT("[Transition] Faction voting subsystem unavailable for Shop->Invader election finalize."));
186 return;
187 }
188
189 FGameplayTag WinnerFactionTag;
190 FGameplayTagContainer WinnerEffectTags;
191 if (!VotingSubsystem->FinalizeElection(WinnerFactionTag, WinnerEffectTags, true, true))
192 {
193 UE_LOG(ARLog, Verbose, TEXT("[Transition] No faction election winner finalized for this Shop->Invader transition."));
194 return;
195 }
196
197 bFactionElectionFinalizedForThisTransition = true;
198 UE_LOG(
199 ARLog,
200 Log,
201 TEXT("[Transition] Finalized faction election winner '%s' for Shop->Invader transition."),
202 *WinnerFactionTag.ToString());

257bool AARTransitionGameMode::TryAdvanceToDestination()
258{
259 if (!HasAuthority() || !bAutoAdvanceWhenAllPlayersReady || bTransitionTravelStarted)
260 {
261 return false;
262 }
263
264 AARTransitionGameState* TransitionGameState = GetGameState<AARTransitionGameState>();
265 if (!TransitionGameState)
266 {
267 return false;
268 }
269
270 if (!TransitionGameState->AreAllPlayersTravelReady())
271 {
272 return false;
273 }
274
275 const FARTransitionContext& TransitionContext = TransitionGameState->GetTransitionContext();
276 if (TransitionContext.DestinationURL.IsEmpty())
277 {
278 UE_LOG(ARLog, Warning, TEXT("[Transition] Cannot advance: destination URL is empty."));
279 return false;
280 }
281
282 bTransitionTravelStarted = true;
283 const FString DestinationTravelURL = ARTransition::AppendTransitionContextOptions(TransitionContext.DestinationURL, TransitionContext);
284 // Use absolute travel for the final transition hop so previous map URL options
285 // (for example a transient game-mode override) do not leak into destination mode.
286 if (!TryStartTravel(DestinationTravelURL, TEXT(""), false, true, false, false, EARTravelRoutePolicy::ForceDirect))
287 {
288 bTransitionTravelStarted = false;
289 UE_LOG(ARLog, Warning, TEXT("[Transition] Advance travel failed to '%s'."), *DestinationTravelURL);
290 return false;
291 }
292
293 UE_LOG(ARLog, Log, TEXT("[Transition] Advancing to destination '%s'."), *DestinationTravelURL);
294 return true;
