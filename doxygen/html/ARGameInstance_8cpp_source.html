164bool UARGameInstance::TryResolveDebugVerbosityArg(const TArray<FString>& Args, FString& OutTargetVerbosity) const
165{
166 const FString LevelArg = (Args.Num() > 0 ? Args[0] : FString()).TrimStartAndEnd().ToLower();
167 if (LevelArg == TEXT("veryverbose") || LevelArg == TEXT("vv"))
168 {
169 OutTargetVerbosity = TEXT("veryverbose");
170 }
171 else if (LevelArg == TEXT("verbose") || LevelArg == TEXT("v"))
172 {
173 OutTargetVerbosity = TEXT("verbose");
174 }
175 else if (LevelArg == TEXT("log") || LevelArg == TEXT("l") || LevelArg == TEXT("default"))
176 {
177 OutTargetVerbosity = TEXT("log");
178 }
179 else if (LevelArg == TEXT("warning") || LevelArg == TEXT("warn") || LevelArg == TEXT("w"))
180 {
181 OutTargetVerbosity = TEXT("warning");
182 }
183 else if (LevelArg == TEXT("error") || LevelArg == TEXT("e"))
184 {
185 OutTargetVerbosity = TEXT("error");
186 }
187 else if (LevelArg == TEXT("off") || LevelArg == TEXT("none"))
188 {
189 OutTargetVerbosity = TEXT("off");
190 }
191 else if (LevelArg == TEXT("reset"))
192 {
193 OutTargetVerbosity = TEXT("log");
194 }
195 else
196 {
197 return false;
198 }
199
200 return true;
201}

203void UARGameInstance::HandleConsoleArDebug(const TArray<FString>& Args)
204{
205 FString TargetVerbosity;
206 if (!TryResolveDebugVerbosityArg(Args, TargetVerbosity))
207 {
208 UE_LOG(ARLog, Log, TEXT("[Debug] Usage: ar.debug.log <veryverbose|verbose|log|warning|error|off|reset>"));
209 return;
210 }
211
212 UWorld* World = GetWorld();
213 bool bApplied = false;
214 if (GEngine && World)
215 {
216 const FString ExecCommand = FString::Printf(TEXT("log arlog %s"), *TargetVerbosity);
217 bApplied = GEngine->Exec(World, *ExecCommand);
218 }
219
220 if (bApplied)
221 {
222 UE_LOG(ARLog, Log, TEXT("[Debug] ARLog verbosity set to '%s' via ar.debug.log."), *TargetVerbosity);
223 }
224 else
225 {
226 UE_LOG(ARLog, Warning, TEXT("[Debug] ar.debug.log could not apply verbosity '%s' (no active world/console context)."), *TargetVerbosity);
227 }
228}

230void UARGameInstance::HandleConsoleArDebugAll(const TArray<FString>& Args)
231{
232 FString TargetVerbosity;
233 if (!TryResolveDebugVerbosityArg(Args, TargetVerbosity))
234 {
235 UE_LOG(ARLog, Log, TEXT("[Debug] Usage: ar.debug.log.all <veryverbose|verbose|log|warning|error|off|reset>"));
236 return;
237 }
238
239 UWorld* World = GetWorld();
240 if (!GEngine || !World)
241 {
242 UE_LOG(ARLog, Warning, TEXT("[Debug] ar.debug.log.all could not apply verbosity '%s' (no active world/console context)."), *TargetVerbosity);
243 return;
244 }
245
246 const TCHAR* Categories[] =
247 {
248 TEXT("arlog"),
249 TEXT("emolog"),
250 TEXT("parleylog"),
251 TEXT("logtagkey")
252 };
253
254 bool bAppliedAll = true;
255 for (const TCHAR* CategoryName : Categories)
256 {
257 const FString ExecCommand = FString::Printf(TEXT("log %s %s"), CategoryName, *TargetVerbosity);
258 bAppliedAll &= GEngine->Exec(World, *ExecCommand);
259 }
260
261 if (bAppliedAll)
262 {
263 UE_LOG(ARLog, Log, TEXT("[Debug] ar.debug.log.all set ARLog, EmoLog, ParleyLog, and TagKeyLog to '%s'."), *TargetVerbosity);
264 }
265 else
266 {
267 UE_LOG(ARLog, Warning, TEXT("[Debug] ar.debug.log.all only partially applied verbosity '%s'."), *TargetVerbosity);
268 }
