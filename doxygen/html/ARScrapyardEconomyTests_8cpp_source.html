41 UWorld* TestWorld = ResolveAutomationWorld_Scrapyard();

47 FActorSpawnParameters SpawnParams;

49
50 AARScrapyardGameState* ScrapyardGameState = TestWorld->SpawnActor<AARScrapyardGameState>(
51 AARScrapyardGameState::StaticClass(),
52 FVector::ZeroVector,
53 FRotator::ZeroRotator,
54 SpawnParams);
55 AARCarryItemBase* ItemActor = TestWorld->SpawnActor<AARCarryItemBase>(
56 AARCarryItemBase::StaticClass(),
57 FVector::ZeroVector,
58 FRotator::ZeroRotator,
59 SpawnParams);
60
61 if (!TestNotNull(TEXT("Spawned scrapyard game state"), ScrapyardGameState)
62 || !TestNotNull(TEXT("Spawned scrapyard carry item"), ItemActor))
63 {
64 if (ItemActor)
65 {
66 ItemActor->Destroy();
67 }
68 if (ScrapyardGameState)
69 {
70 ScrapyardGameState->Destroy();

75 FIntProperty* FallbackCostProperty = FindFProperty<FIntProperty>(AARCarryItemBase::StaticClass(), TEXT("FallbackScrapCost"));

77 {
78 ItemActor->Destroy();
79 ScrapyardGameState->Destroy();
80 return false;
81 }
82 FallbackCostProperty->SetPropertyValue_InContainer(ItemActor, 12);
83
84 ScrapyardGameState->SetScrapFromSave(0);
85
86 int32 ReservedCost = 0;
87 TestTrue(TEXT("Reserve succeeds for carry item"), ScrapyardGameState->ReserveScrapForItem(ItemActor, ReservedCost));

90 TestEqual(TEXT("Extraction summary tracks reserved cost"), ScrapyardGameState->GetExtractionSummary().ReservedCostTotal, 12);

92
93 int32 RefundCost = 0;
94 TestTrue(TEXT("Refund succeeds for previously reserved item"), ScrapyardGameState->RefundScrapForItem(ItemActor, RefundCost));
95 TestEqual(TEXT("Refund cost equals reserved cost"), RefundCost, 12);
96 TestEqual(TEXT("Scrap refund returns balance to zero"), ScrapyardGameState->GetScrap(), 0);
97 TestEqual(TEXT("Reserved cost clears after refund"), ScrapyardGameState->GetExtractionSummary().ReservedCostTotal, 0);
98 TestEqual(TEXT("Reserved item count clears after refund"), ScrapyardGameState->GetExtractionSummary().ReservedItemCount, 0);
99
100 ItemActor->Destroy();
101 ScrapyardGameState->Destroy();
102 return true;
103}
