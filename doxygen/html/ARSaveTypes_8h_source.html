52USTRUCT(BlueprintType)
53struct ALIENRAMEN_API FARMeatState
54{
55 GENERATED_BODY()
56
57 // Canonical tuple-backed meat inventory entries.
58 // Entries are normalized/sorted by MeatType + MeatColor + MeatQualityTier.
59 UPROPERTY(EditAnywhere, BlueprintReadWrite, Category = "Save")
60 TArray<FARMeatTypeAmount> AdditionalAmountsByType;
61
62 int32 GetTotalAmount() const
63 {
64 int32 Total = 0;
65 for (const FARMeatTypeAmount& Entry : AdditionalAmountsByType)
66 {
67 Total += FMath::Max(0, Entry.Amount);
68 }
69 return Total;
70 }
71
72 void NormalizeAdditionalAmounts()
73 {
74 TMap<FString, FARMeatTypeAmount> Aggregated;
75 for (const FARMeatTypeAmount& Entry : AdditionalAmountsByType)
76 {
77 if (!Entry.MeatType.IsValid())
78 {
79 continue;
80 }
81
82 const int32 SanitizedAmount = FMath::Max(0, Entry.Amount);
83 if (SanitizedAmount <= 0)
84 {
85 continue;
86 }
87
88 FARMeatTypeAmount SanitizedEntry = Entry;
89 SanitizedEntry.Amount = SanitizedAmount;
90 SanitizedEntry.MeatColor = SanitizedEntry.MeatColor == EARAffinityColor::Unknown ? EARAffinityColor::None : SanitizedEntry.MeatColor;
91 if (!StaticEnum<EARVendingQualityTier>()->IsValidEnumValue(static_cast<int64>(SanitizedEntry.MeatQualityTier)))
92 {
93 SanitizedEntry.MeatQualityTier = EARVendingQualityTier::Standard;
94 }
95
96 const FString Key = FString::Printf(
97 TEXT("%s|%d|%d"),
98 *SanitizedEntry.MeatType.ToString(),
99 static_cast<int32>(SanitizedEntry.MeatColor),
100 static_cast<int32>(SanitizedEntry.MeatQualityTier));
101 FARMeatTypeAmount& AggregatedEntry = Aggregated.FindOrAdd(Key);
102 if (!AggregatedEntry.MeatType.IsValid())
103 {
104 AggregatedEntry.MeatType = SanitizedEntry.MeatType;
105 AggregatedEntry.MeatColor = SanitizedEntry.MeatColor;
106 AggregatedEntry.MeatQualityTier = SanitizedEntry.MeatQualityTier;
107 }
108
109 AggregatedEntry.Amount += SanitizedAmount;
110 }
111
112 AdditionalAmountsByType.Reset(Aggregated.Num());
113 for (const TPair<FString, FARMeatTypeAmount>& Pair : Aggregated)
114 {
115 AdditionalAmountsByType.Add(Pair.Value);
116 }
117
118 AdditionalAmountsByType.Sort([](const FARMeatTypeAmount& A, const FARMeatTypeAmount& B)
119 {
120 const FString AType = A.MeatType.ToString();
121 const FString BType = B.MeatType.ToString();
122 if (AType != BType)
123 {
124 return AType < BType;
125 }
126
127 const int32 AColor = static_cast<int32>(A.MeatColor);
128 const int32 BColor = static_cast<int32>(B.MeatColor);
129 if (AColor != BColor)
130 {
131 return AColor < BColor;
132 }
133
134 return static_cast<int32>(A.MeatQualityTier) < static_cast<int32>(B.MeatQualityTier);
135 });
136 }

176 bool Matches(const FARPlayerIdentity& Other) const
177 {
178 const bool bHasStrictIdentity = HasStrictOnlineIdentity();
179 const bool bOtherHasStrictIdentity = Other.HasStrictOnlineIdentity();
180 if (bHasStrictIdentity || bOtherHasStrictIdentity)
181 {
182 if (!bHasStrictIdentity || !bOtherHasStrictIdentity)
183 {
184 return false;
185 }
186
187 if (!UniqueNetIdType.Equals(Other.UniqueNetIdType, ESearchCase::CaseSensitive))
188 {
189 return false;
190 }
191
192 if (!UniqueNetIdString.Equals(Other.UniqueNetIdString, ESearchCase::CaseSensitive))
193 {
194 return false;
195 }
196
197 return bSharedOnlineIdSecondaryProfile == Other.bSharedOnlineIdSecondaryProfile;
198 }
199
200 const FString ThisDisplayName = DisplayName.ToString().TrimStartAndEnd();
201 const FString OtherDisplayName = Other.DisplayName.ToString().TrimStartAndEnd();
202 if (ThisDisplayName.IsEmpty() || OtherDisplayName.IsEmpty())
203 {
204 return false;
205 }
206
207 return bSharedOnlineIdSecondaryProfile == Other.bSharedOnlineIdSecondaryProfile
208 && ThisDisplayName.Equals(OtherDisplayName, ESearchCase::CaseSensitive);
