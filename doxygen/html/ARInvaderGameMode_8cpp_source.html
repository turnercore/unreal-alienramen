99bool AARInvaderGameMode::FinalizeInvaderRunAndTravel(const FString& InTravelURL)
100{
101 if (!HasAuthority())
102 {
103 UE_LOG(ARLog, Warning, TEXT("[InvaderGameMode] FinalizeInvaderRunAndTravel ignored: not authority."));
104 return false;
105 }
106
107 const FString TravelURL = InTravelURL.IsEmpty() ? DefaultPostRunTravelURL : InTravelURL;
108 if (TravelURL.IsEmpty())
109 {
110 UE_LOG(ARLog, Warning, TEXT("[InvaderGameMode] FinalizeInvaderRunAndTravel failed: destination URL is empty."));
111 return false;
112 }
113
114 if (UWorld* World = GetWorld())
115 {
116 World->GetTimerManager().ClearTimer(RunEndAutoTravelTimer);
117 RunEndAutoTravelTimer.Invalidate();
118 }
119
120 if (bRunEndTravelRequested)
121 {
122 UE_LOG(ARLog, Verbose, TEXT("[InvaderGameMode] FinalizeInvaderRunAndTravel ignored: travel already requested."));
123 return true;
124 }
125
126 bRunEndTravelRequested = true;
127 if (!EndModeAndTravel(TravelURL, TEXT(""), true))
128 {
129 bRunEndTravelRequested = false;
130 UE_LOG(ARLog, Warning, TEXT("[InvaderGameMode] FinalizeInvaderRunAndTravel failed for URL '%s'."), *TravelURL);
131 return false;
132 }
133
134 UE_LOG(ARLog, Log, TEXT("[InvaderGameMode] FinalizeInvaderRunAndTravel started: URL='%s' Reason=%d."),
135 *TravelURL,
136 static_cast<int32>(LastHandledRunEndReason));
137 return true;
138}

261bool AARInvaderGameMode::ResolveInvaderPawnClassFromShipTag(const FGameplayTag ShipTag, TSubclassOf<APawn>& OutPawnClass) const
262{
263 OutPawnClass = nullptr;
264 if (!ShipTag.IsValid())
265 {
266 return false;
267 }
268
269 UGameInstance* GameInstance = GetGameInstance();
270 UTagKeySubsystem* Resolver = GameInstance ? GameInstance->GetSubsystem<UTagKeySubsystem>() : nullptr;
271 if (!Resolver)
272 {
273 return false;
274 }
275
276 FInstancedStruct ShipRow;
277 FString ResolveError;
278 if (!Resolver->TryResolveRowStructForTag(ShipTag, ShipRow, ResolveError))
279 {
280 UE_LOG(
281 ARLog,
282 Warning,
283 TEXT("[InvaderGameMode] Failed resolving ship row '%s': %s"),
284 *ShipTag.ToString(),
285 *ResolveError);
286 return false;
287 }
288
289 const FARShipDefRow* ShipDef = ShipRow.GetPtr<FARShipDefRow>();
290 if (!ShipDef)
291 {
292 const UScriptStruct* RowStruct = ShipRow.GetScriptStruct();
293 UE_LOG(
294 ARLog,
295 Error,
296 TEXT("[InvaderGameMode] Ship row '%s' resolved to unexpected struct '%s'; expected FARShipDefRow."),
297 *ShipTag.ToString(),
298 *GetNameSafe(RowStruct));
299 return false;
300 }
301
302 if (ShipDef->InvaderPawnClass.IsNull())
303 {
304 return false;
305 }
306
307 if (UClass* PawnClass = ShipDef->InvaderPawnClass.LoadSynchronous())
308 {
309 OutPawnClass = PawnClass;
310 return true;
311 }
312
313 UE_LOG(
314 ARLog,
315 Error,
316 TEXT("[InvaderGameMode] Ship row '%s' InvaderPawnClass failed to load (Path=%s)."),
317 *ShipTag.ToString(),
318 *ShipDef->InvaderPawnClass.ToString());
319 return false;
320}

322bool AARInvaderGameMode::FindFirstTagUnderRoot(const FGameplayTagContainer& InTags, const FGameplayTag& RootTag, FGameplayTag& OutTag)
323{
324 OutTag = FGameplayTag();
325 if (!RootTag.IsValid())
326 {
327 return false;
328 }
329
330 for (const FGameplayTag Tag : InTags)

338
