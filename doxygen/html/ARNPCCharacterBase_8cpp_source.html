131void AARNPCCharacterBase::ResolveOptionalComponents()
132{
133 TArray<UParleySpeakerComponent*> TalkComponents;
134 GetComponents(TalkComponents);
135 if (!TalkComponents.IsEmpty())
136 {
137 UParleySpeakerComponent* PreferredTalkComponent = SpeakerComponent;
138 if (!PreferredTalkComponent || !TalkComponents.Contains(PreferredTalkComponent))
139 {
140 PreferredTalkComponent = TalkComponents[0];
141 }
142
143 if (PreferredTalkComponent && !PreferredTalkComponent->GetSpeakerTag().IsValid())
144 {
145 for (UParleySpeakerComponent* Candidate : TalkComponents)
146 {
147 if (Candidate && Candidate->GetSpeakerTag().IsValid())
148 {
149 PreferredTalkComponent = Candidate;
150 break;
151 }
152 }
153 }
154
155 SpeakerComponent = PreferredTalkComponent;
156 if (TalkComponents.Num() > 1)
157 {
158 UE_LOG(
159 ARLog,
160 Warning,
161 TEXT("[Speaker] '%s' has %d UParleySpeakerComponent instances. Using '%s' as canonical."),
162 *GetNameSafe(this),
163 TalkComponents.Num(),
164 *GetNameSafe(SpeakerComponent));
165 }
166 }
167 else
168 {
169 SpeakerComponent = nullptr;
170 }
171
172 TArray<UEmoComponent*> EmotionComponents;
173 GetComponents(EmotionComponents);
174 if (!EmotionComponents.IsEmpty())
175 {
176 UEmoComponent* PreferredEmotionComponent = EmotionComponent;
177 if (!PreferredEmotionComponent || !EmotionComponents.Contains(PreferredEmotionComponent))
178 {
179 PreferredEmotionComponent = EmotionComponents[0];
180 }
181
182 EmotionComponent = PreferredEmotionComponent;
183 if (EmotionComponents.Num() > 1)
184 {
185 UE_LOG(
186 ARLog,
187 Warning,
188 TEXT("[Speaker] '%s' has %d UEmoComponent instances. Using '%s' as canonical."),
189 *GetNameSafe(this),
190 EmotionComponents.Num(),
191 *GetNameSafe(EmotionComponent));
192 }
193 }
194 else
195 {
196 EmotionComponent = nullptr;
197 }
198
199 TArray<UARCustomerComponent*> CustomerComponents;
200 GetComponents(CustomerComponents);
201 if (!CustomerComponents.IsEmpty())
202 {
203 UARCustomerComponent* PreferredCustomerComponent = CustomerComponent;
204 if (!PreferredCustomerComponent || !CustomerComponents.Contains(PreferredCustomerComponent))
205 {
206 PreferredCustomerComponent = CustomerComponents[0];
207 }
208
209 CustomerComponent = PreferredCustomerComponent;
210 if (CustomerComponents.Num() > 1)
211 {
212 UE_LOG(
213 ARLog,
214 Warning,
215 TEXT("[Speaker] '%s' has %d UARCustomerComponent instances. Using '%s' as canonical."),
216 *GetNameSafe(this),
217 CustomerComponents.Num(),
218 *GetNameSafe(CustomerComponent));
219 }
220 }
221 else
222 {
223 CustomerComponent = nullptr;
224 }
225}

284void AARNPCCharacterBase::InteractByController(AARPlayerController* InteractingController)
285{
286 if (!InteractingController)
287 {
288 UE_LOG(ARLog, Warning, TEXT("[Interact] '%s' interaction ignored: InteractingController is null."), *GetNameSafe(this));
289 return;
290 }
291
292 // Runtime resilience: resolve optional components at interaction time in case this actor was reinstanced
293 // or component pointers became stale after hot reload/editor world transitions.
294 if (!SpeakerComponent || !CustomerComponent || !EmotionComponent)
295 {
296 ResolveOptionalComponents();
297 RefreshStateTreeInteractionFlags();
298 }
299
300 const bool bHasActiveCustomerOrder = CustomerComponent && CustomerComponent->HasActiveOrder();
301 if (HasAuthority() && CustomerComponent)
302 {
303 FARRamenServeResult ServeResult;
304 if (CustomerComponent->TryServeHeldBowlFromController(InteractingController, ServeResult))
305 {
306 return;
307 }
308
309 if (bHasActiveCustomerOrder)
310 {
311 UE_LOG(
312 ARLog,
313 Verbose,
314 TEXT("[Interact] '%s' has active customer order but no valid bowl was served by '%s'; trying speaker fallback."),
315 *GetNameSafe(this),
316 *GetNameSafe(InteractingController));
317 }
318 }
319
320 if (!SpeakerComponent)
321 {
322 if (HasAuthority() && CustomerComponent)
323 {
324 const FGameplayTag CustomerSpeakerTag = CustomerComponent->GetSpeakerTag();
325 if (CustomerSpeakerTag.IsValid())
326 {
327 if (UGameInstance* GameInstance = GetGameInstance())
328 {
329 if (UParleyDialogueSubsystem* DialogueSubsystem = GameInstance->GetSubsystem<UParleyDialogueSubsystem>())
330 {
331 FGameplayTag SourceSpeakerTag;
332 if (const APawn* InteractingPawn = InteractingController->GetPawn())
333 {
334 if (const UParleySpeakerComponent* SourceSpeakerComponent = InteractingPawn->FindComponentByClass<UParleySpeakerComponent>())
335 {
336 SourceSpeakerTag = SourceSpeakerComponent->GetSpeakerTag();
337 }
338 }
339 if (!SourceSpeakerTag.IsValid())
340 {
341 UE_LOG(
342 ARLog,
343 Verbose,
344 TEXT("[Interact] '%s' cannot start dialogue via customer speaker '%s': controller '%s' has no possessed pawn speaker component."),
345 *GetNameSafe(this),
346 *CustomerSpeakerTag.ToString(),
347 *GetNameSafe(InteractingController));
348 return;
349 }
350
351 if (DialogueSubsystem->TryStartDialogueBetweenSpeakers(InteractingController, SourceSpeakerTag, CustomerSpeakerTag))
352 {
353 UE_LOG(
354 ARLog,
355 Verbose,
356 TEXT("[Interact] '%s' started dialogue via customer speaker '%s'."),
357 *GetNameSafe(this),
358 *CustomerSpeakerTag.ToString());
359 return;
360 }
361 }
362 }
363 }
364 }
365
366 LogMissingSpeakerComponentDiagnostics(this);
367 return;
368 }
369
370 if (!bSpeakerLocalStateAllowsDialogue && !bHasActiveCustomerOrder)
371 {
372 UE_LOG(ARLog, Verbose, TEXT("[Speaker] Interact ignored for '%s': local state blocks dialogue."), *GetNameSafe(this));
373 return;
374 }
375
376 SpeakerComponent->InteractByController(InteractingController);
377}

576void AARNPCCharacterBase::RefreshAutoWantsToTalkEmotion(const bool bEffectiveTalkable)
577{
578 if (!HasAuthority() || !EmotionComponent)
579 {
580 UE_LOG(
581 ARLog,
582 Verbose,
583 TEXT("[Emotion][Talkable] Skip auto wants-to-talk for '%s': Authority=%s EmotionComponent=%s"),
584 *GetNameSafe(this),
585 HasAuthority() ? TEXT("true") : TEXT("false"),
586 EmotionComponent ? TEXT("valid") : TEXT("null"));
587 return;
588 }
589
590 const UEmoSettings* EmotionSettings = GetDefault<UEmoSettings>();
591 const FGameplayTag WantsToTalkTag = EmotionSettings ? EmotionSettings->WantsToTalkEmotionTag : FGameplayTag();
592 if (!WantsToTalkTag.IsValid())
593 {
594 EmotionComponent->ClearEmotionRegistration(TalkableStateEmotionSourceId);
595 bAutoWantsToTalkEmotionApplied = false;
596 UE_LOG(ARLog, Verbose, TEXT("[Emotion][Talkable] '%s' no WantsToTalkEmotionTag configured; cleared TalkableState source."), *GetNameSafe(this));
597 return;
598 }
599
600 if (bEffectiveTalkable)
601 {
602 EmotionComponent->SetEmotionRegistration(TalkableStateEmotionSourceId, WantsToTalkTag, 0);
603 bAutoWantsToTalkEmotionApplied = true;
604 UE_LOG(
605 ARLog,
606 Verbose,
607 TEXT("[Emotion][Talkable] '%s' apply TalkableState=%s (effective talkable=true)."),
608 *GetNameSafe(this),
609 *WantsToTalkTag.ToString());
610 return;
611 }
612
613 EmotionComponent->ClearEmotionRegistration(TalkableStateEmotionSourceId);
614 UE_LOG(ARLog, Verbose, TEXT("[Emotion][Talkable] '%s' cleared TalkableState (effective talkable=false)."), *GetNameSafe(this));
615 bAutoWantsToTalkEmotionApplied = false;
616}
